# T4.4 — Web UI Core Pages: Dashboard, Transactions, Accounts

## Goal

Implement the Dashboard, Transactions, and Accounts pages in the React web UI
with real data fetched from the API.

## Context

These three pages cover the main data browsing use case. The Dashboard gives a quick
financial snapshot; Transactions lists the register; Accounts shows balances.

**Depends on:** T4.3 (web setup), T4.2 (API)

### Implementation

```tsx
import { useQuery } from "@tanstack/react-query";
import { api } from "../api/client";
import {
  BarChart,
  Bar,
  XAxis,
  YAxis,
  Tooltip,
  ResponsiveContainer,
} from "recharts";

export default function Dashboard() {
  const { data: accounts } = useQuery({
    queryKey: ["accounts"],
    queryFn: api.getAccounts,
  });

  const { data: recent } = useQuery({
    queryKey: ["transactions", { limit: "10" }],
    queryFn: () => api.getTransactions({ limit: "10" }),
  });

  const { data: is } = useQuery({
    queryKey: ["income-statement", "current-month"],
    queryFn: () => {
      const now = new Date();
      const begin = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}-01`;
      return api.getIncomeStatement({ begin, period: "monthly" });
    },
  });

  // Calculate net worth: sum of all asset accounts
  const netWorth =
    accounts?.accounts
      .filter((a) => a.type === "asset")
      .reduce((sum, a) => sum + a.balance, 0) ?? 0;

  // Spending by category data for chart
  const spendingData = Object.entries(is?.expenses ?? {})
    .map(([account, amount]) => ({
      name: account.replace("expenses:", ""),
      amount: Math.abs(amount),
    }))
    .sort((a, b) => b.amount - a.amount)
    .slice(0, 8);

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold text-gray-900">Dashboard</h1>

      {/* Summary widgets */}
      <div className="grid grid-cols-3 gap-4">
        <StatCard title="Net Worth" value={formatCurrency(netWorth)} />
        <StatCard
          title="Monthly Income"
          value={formatCurrency(totalRevenues(is))}
          positive
        />
        <StatCard
          title="Monthly Expenses"
          value={formatCurrency(totalExpenses(is))}
          negative
        />
      </div>

      {/* Spending chart */}
      <div className="bg-white rounded-lg shadow p-4">
        <h2 className="text-lg font-semibold mb-4">Spending by Category</h2>
        <ResponsiveContainer width="100%" height={250}>
          <BarChart data={spendingData}>
            <XAxis dataKey="name" />
            <YAxis />
            <Tooltip formatter={(v) => formatCurrency(v as number)} />
            <Bar dataKey="amount" fill="#7C3AED" />
          </BarChart>
        </ResponsiveContainer>
      </div>

      {/* Recent transactions */}
      <div className="bg-white rounded-lg shadow">
        <div className="p-4 border-b">
          <h2 className="text-lg font-semibold">Recent Transactions</h2>
        </div>
        <TransactionTable transactions={recent?.transactions ?? []} />
      </div>
    </div>
  );
}
```

### Helper components in `web-ui/src/components/`:

`StatCard.tsx`:

```tsx
interface Props {
  title: string;
  value: string;
  positive?: boolean;
  negative?: boolean;
}
export function StatCard({ title, value, positive, negative }: Props) {
  const valueClass = positive
    ? "text-green-600"
    : negative
      ? "text-red-600"
      : "text-gray-900";
  return (
    <div className="bg-white rounded-lg shadow p-4">
      <div className="text-sm text-gray-500">{title}</div>
      <div className={`text-2xl font-bold ${valueClass}`}>{value}</div>
    </div>
  );
}
```

`TransactionTable.tsx`:

```tsx
interface Props {
  transactions: Transaction[];
}
export function TransactionTable({ transactions }: Props) {
  return (
    <table className="w-full text-sm">
      <thead className="bg-gray-50 text-gray-500 uppercase text-xs">
        <tr>
          <th className="px-4 py-2 text-left">Date</th>
          <th className="px-4 py-2 text-left">Description</th>
          <th className="px-4 py-2 text-left">Account</th>
          <th className="px-4 py-2 text-right">Amount</th>
        </tr>
      </thead>
      <tbody className="divide-y divide-gray-100">
        {transactions.flatMap((txn) =>
          txn.postings.map((posting, i) => (
            <tr key={`${txn.id}-${i}`} className="hover:bg-gray-50">
              <td className="px-4 py-2 text-gray-500">
                {i === 0 ? txn.date : ""}
              </td>
              <td className="px-4 py-2">{i === 0 ? txn.description : ""}</td>
              <td className="px-4 py-2 text-blue-600 font-mono text-xs">
                {posting.account}
              </td>
              <td
                className={`px-4 py-2 text-right font-mono ${posting.amount >= 0 ? "text-gray-900" : "text-red-600"}`}
              >
                {formatCurrency(posting.amount, posting.currency)}
              </td>
            </tr>
          )),
        )}
      </tbody>
    </table>
  );
}
```

`formatCurrency.ts`:

```typescript
export function formatCurrency(amount: number, currency = "USD"): string {
  const symbols: Record<string, string> = { USD: "$", GBP: "£", EUR: "€" };
  const sym = symbols[currency] ?? "";
  const formatted = Math.abs(amount).toLocaleString("en-US", {
    minimumFractionDigits: 2,
    maximumFractionDigits: 2,
  });
  return `${amount < 0 ? "-" : ""}${sym}${formatted}${sym ? "" : " " + currency}`;
}
```

## `web-ui/src/pages/Transactions.tsx`

### Features

- Search input (filters by description)
- Date range inputs (begin/end)
- Account filter dropdown
- Paginated table (25 per page, LIMIT/OFFSET)
- Each row: date, description, postings (account + amount per posting)

```tsx
export default function Transactions() {
  const [search, setSearch] = useState("");
  const [begin, setBegin] = useState("");
  const [end, setEnd] = useState("");
  const [page, setPage] = useState(0);
  const limit = 25;

  const { data, isLoading } = useQuery({
    queryKey: ["transactions", { search, begin, end, page }],
    queryFn: () =>
      api.getTransactions({
        desc: search,
        begin,
        end,
        limit: String(limit),
        offset: String(page * limit),
      }),
  });

  return (
    <div className="space-y-4">
      <h1 className="text-2xl font-bold">Transactions</h1>

      {/* Filters */}
      <div className="flex gap-3 bg-white p-3 rounded-lg shadow">
        <input
          placeholder="Search description..."
          value={search}
          onChange={(e) => setSearch(e.target.value)}
          className="..."
        />
        <input
          type="date"
          value={begin}
          onChange={(e) => setBegin(e.target.value)}
          className="..."
        />
        <input
          type="date"
          value={end}
          onChange={(e) => setEnd(e.target.value)}
          className="..."
        />
      </div>

      {/* Table */}
      {isLoading ? (
        <LoadingSpinner />
      ) : (
        <>
          <TransactionTable transactions={data?.transactions ?? []} />
          <Pagination
            page={page}
            total={data?.total ?? 0}
            limit={limit}
            onPageChange={setPage}
          />
        </>
      )}
    </div>
  );
}
```

## `web-ui/src/pages/Accounts.tsx`

### Features

- Account balance list grouped by type (assets, liabilities, income, expenses)
- Each account shows: name, balance, transaction count
- Click on account → filters Transactions page to that account

```tsx
export default function Accounts() {
  const { data } = useQuery({
    queryKey: ["accounts"],
    queryFn: api.getAccounts,
  });

  const grouped = groupBy(data?.accounts ?? [], (a) => a.type);
  const types = ["asset", "liability", "equity", "income", "expense"] as const;

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold">Accounts</h1>
      {types.map((type) => {
        const accts = grouped[type] ?? [];
        if (!accts.length) return null;
        return (
          <div key={type} className="bg-white rounded-lg shadow">
            <div className="p-4 border-b capitalize font-semibold">{type}s</div>
            <table className="w-full text-sm">
              <thead>...</thead>
              <tbody>
                {accts.map((a) => (
                  <tr key={a.name}>
                    <td className="px-4 py-2 font-mono text-blue-600">
                      {a.name}
                    </td>
                    <td className="px-4 py-2 text-right">
                      {formatCurrency(a.balance, a.currency)}
                    </td>
                    <td className="px-4 py-2 text-right text-gray-500">
                      {a.transaction_count} txns
                    </td>
                  </tr>
                ))}
              </tbody>
            </table>
          </div>
        );
      })}
    </div>
  );
}
```
