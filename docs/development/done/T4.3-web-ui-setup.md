# T4.3 — React Web UI Setup

## Goal

Set up the React + Vite + TypeScript + Tailwind CSS frontend project structure,
configure the build pipeline so it outputs into the Go embed target directory,
and implement the basic layout shell (navigation sidebar, page routing).

## Context

The web interface runs at port 4444 and serves compiled React assets embedded in
the Go binary. This task sets up the frontend project and the Go embedding layer,
leaving actual page content for T4.4 and T4.5.

**Depends on:** T4.2 (API endpoints that the web UI will call)
**Blocks:** T4.4, T4.5 (page content)

### `web-ui/src/api/client.ts` — API client

```typescript
const BASE_URL = "/api";

async function fetchJSON<T>(path: string, options?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    headers: { "Content-Type": "application/json" },
    ...options,
  });
  if (!res.ok) {
    const err = await res.json().catch(() => ({ error: res.statusText }));
    throw new Error(err.error || res.statusText);
  }
  return res.json();
}

export const api = {
  getTransactions: (params?: Record<string, string>) =>
    fetchJSON<{ transactions: Transaction[]; total: number }>(
      "/transactions" + (params ? "?" + new URLSearchParams(params) : ""),
    ),
  getAccounts: () => fetchJSON<{ accounts: Account[] }>("/accounts"),
  getBalance: (params?: Record<string, string>) =>
    fetchJSON<{ accounts: Account[]; total: number }>("/reports/balance"),
  getIncomeStatement: (params?: Record<string, string>) =>
    fetchJSON<IncomeStatement>("/reports/income-statement"),
  runFQL: (query: string) =>
    fetchJSON<FQLResult>("/fql", {
      method: "POST",
      body: JSON.stringify({ query }),
    }),
};
```

## Create `web/server.go` — Go Web Server with go:embed

```go
package web

import (
    "embed"
    "io/fs"
    "net/http"
)

//go:embed static/dist
var staticFiles embed.FS

// Handler returns an http.Handler that serves the embedded React app.
// All routes not matching /api are served as the SPA (index.html).
func Handler() http.Handler {
    dist, err := fs.Sub(staticFiles, "static/dist")
    if err != nil {
        panic("web: could not sub static files: " + err.Error())
    }

    fileServer := http.FileServer(http.FS(dist))

    return http.HandlerFunc(func(w http.ResponseWriter, r *http.Request) {
        // Try to serve the file directly
        // If it doesn't exist (404), serve index.html (SPA routing)
        _, err := dist.Open(r.URL.Path[1:]) // strip leading /
        if err != nil {
            // Serve index.html for SPA routing
            r.URL.Path = "/"
        }
        fileServer.ServeHTTP(w, r)
    })
}
```

	./doublebook api

- [ ] Navigating between pages works (React Router)
