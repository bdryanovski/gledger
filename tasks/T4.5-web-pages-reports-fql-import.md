### Income Statement tab

Display the income statement as a formatted table:

```
Income Statement
Period: [begin date picker] to [end date picker]  [Refresh] [Print] [Export PDF]

Revenues
  income:salary          $2,000.00   40%
  income:freelance         $500.00   10%
  ─────────────────────────────────────────
  Total Revenue          $2,500.00

Expenses
  expenses:housing       $1,200.00   48%
  expenses:food            $200.00    8%
  ─────────────────────────────────────────
  Total Expenses         $1,600.00

Net Income                 $900.00
```

```tsx
export default function Reports() {
  const [tab, setTab] = useState<"income" | "balance">("income");
  const [begin, setBegin] = useState(firstDayOfMonth());
  const [end, setEnd] = useState(today());

  const { data: is } = useQuery({
    queryKey: ["income-statement", begin, end],
    queryFn: () => api.getIncomeStatement({ begin, end }),
    enabled: !!begin && !!end,
  });

  const handlePrint = () => window.print();

  return (
    <div className="space-y-4">
      <div className="flex justify-between items-center">
        <h1 className="text-2xl font-bold">Reports</h1>
        <div className="flex gap-2">
          <button onClick={handlePrint} className="btn-secondary">
            Print
          </button>
        </div>
      </div>

      {/* Tab switcher */}
      <div className="flex border-b">
        <button
          onClick={() => setTab("income")}
          className={tabClass(tab === "income")}
        >
          Income Statement
        </button>
        <button
          onClick={() => setTab("balance")}
          className={tabClass(tab === "balance")}
        >
          Balance Sheet
        </button>
      </div>

      {/* Date range */}
      <div className="flex gap-3 bg-white p-3 rounded-lg shadow">
        <label>
          From:{" "}
          <input
            type="date"
            value={begin}
            onChange={(e) => setBegin(e.target.value)}
          />
        </label>
        <label>
          To:{" "}
          <input
            type="date"
            value={end}
            onChange={(e) => setEnd(e.target.value)}
          />
        </label>
      </div>

      {/* Report content */}
      {tab === "income" ? (
        <IncomeStatementTable data={is} />
      ) : (
        <BalanceSheetTable />
      )}
    </div>
  );
}
```


```tsx
export default function FQL() {
  const [query, setQuery] = useState(
    "SELECT account, SUM(amount) AS total\nFROM accounts\nORDER BY total DESC",
  );
  const [result, setResult] = useState<FQLResult | null>(null);
  const [error, setError] = useState<string | null>(null);
  const [loading, setLoading] = useState(false);
  const [history, setHistory] = useState<string[]>([]);

  const run = async () => {
    setLoading(true);
    setError(null);
    try {
      const r = await api.runFQL(query);
      setResult(r);
      setHistory((h) => [query, ...h.filter((q) => q !== query)].slice(0, 20));
    } catch (e) {
      setError(String(e));
    } finally {
      setLoading(false);
    }
  };

  const onKeyDown = (e: React.KeyboardEvent) => {
    if ((e.ctrlKey || e.metaKey) && e.key === "Enter") run();
  };

  return (
    <div className="space-y-4">
      <h1 className="text-2xl font-bold">FQL Query</h1>

      <div className="bg-white rounded-lg shadow p-4 space-y-3">
        <textarea
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          onKeyDown={onKeyDown}
          rows={5}
          className="w-full font-mono text-sm border rounded p-2 focus:outline-none focus:ring-2 focus:ring-purple-500"
          placeholder="SELECT ... FROM transactions WHERE ..."
        />
        <div className="flex justify-between items-center">
          <div className="text-xs text-gray-500">
            Tables: <code>transactions</code> | <code>accounts</code> |{" "}
            <code>spending</code>
            &nbsp;• Ctrl+Enter to run
          </div>
          <div className="flex gap-2">
            <button
              onClick={() => {
                setQuery("");
                setResult(null);
                setError(null);
              }}
              className="btn-ghost"
            >
              Clear
            </button>
            <button onClick={run} disabled={loading} className="btn-primary">
              {loading ? "Running..." : "Run Query"}
            </button>
          </div>
        </div>
      </div>

      {/* Query history */}
      {history.length > 0 && (
        <details className="text-sm">
          <summary className="cursor-pointer text-gray-500">
            Query history ({history.length})
          </summary>
          <div className="mt-2 space-y-1">
            {history.map((q, i) => (
              <button
                key={i}
                onClick={() => setQuery(q)}
                className="block text-left text-blue-600 hover:underline truncate w-full"
              >
                {q.replace(/\s+/g, " ")}
              </button>
            ))}
          </div>
        </details>
      )}

      {/* Results */}
      {error && (
        <div className="bg-red-50 border border-red-200 rounded p-3 text-red-700 text-sm">
          <strong>Error:</strong> {error}
        </div>
      )}

      {result && (
        <div className="bg-white rounded-lg shadow">
          <div className="p-3 border-b flex justify-between text-sm text-gray-500">
            <span>{result.row_count} rows</span>
            <span>{result.query_ms.toFixed(1)}ms</span>
          </div>
          <FQLResultTable result={result} />
        </div>
      )}
    </div>
  );
}
```

## `web-ui/src/pages/Import.tsx`

### Layout

```
┌─────────────────────────────────────────────────────┐
│  Import CSV                                          │
├─────────────────────────────────────────────────────┤
│  1. Select Import Map                                │
│     [Drag importmap.json here or browse]            │
├─────────────────────────────────────────────────────┤
│  2. Select CSV File                                  │
│     [Drag CSV here or browse]                        │
├─────────────────────────────────────────────────────┤
│  [Dry Run]  [Import]                                 │
├─────────────────────────────────────────────────────┤
│  Results: 89 imported, 63 skipped, 0 errors          │
└─────────────────────────────────────────────────────┘
```

```tsx
export default function Import() {
  const [importMap, setImportMap] = useState<File | null>(null);
  const [csvFile, setCsvFile] = useState<File | null>(null);
  const [result, setResult] = useState<ImportResult | null>(null);
  const [loading, setLoading] = useState(false);

  const doImport = async (dryRun: boolean) => {
    if (!importMap || !csvFile) return;
    setLoading(true);
    const formData = new FormData();
    formData.append("map", importMap);
    formData.append("csv", csvFile);
    formData.append("dry_run", String(dryRun));

    try {
      const res = await fetch("/api/import", {
        method: "POST",
        body: formData,
      });
      setResult(await res.json());
    } finally {
      setLoading(false);
    }
  };

  return (
    <div className="space-y-6">
      <h1 className="text-2xl font-bold">Import CSV</h1>

      <FileDropZone
        label="Import Map (.importmap.json)"
        accept=".json"
        onFile={setImportMap}
      />
      <FileDropZone label="CSV File" accept=".csv" onFile={setCsvFile} />

      <div className="flex gap-3">
        <button
          onClick={() => doImport(true)}
          disabled={!importMap || !csvFile || loading}
          className="btn-secondary"
        >
          Dry Run
        </button>
        <button
          onClick={() => doImport(false)}
          disabled={!importMap || !csvFile || loading}
          className="btn-primary"
        >
          {loading ? "Importing..." : "Import"}
        </button>
      </div>

      {result && <ImportResults result={result} />}
    </div>
  );
}
```

### `FileDropZone` component

A drag-and-drop area with click-to-browse fallback.
