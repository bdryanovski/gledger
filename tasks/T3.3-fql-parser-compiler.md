# T3.3 — FQL Parser and SQL Compiler

## Goal
Build the FQL parser (produces a query AST from FQL tokens) and the SQL compiler
(translates the FQL AST into parameterized SQLite SQL). Fix all the bugs from the
FQL examples (COUNT(*) without alias, HAVING with negative numbers, GROUP BY aggregates).

## Context
This is the core of the FQL system. The parser produces a structured representation
of the query, and the compiler translates it to safe parameterized SQL that can be
executed against the SQLite cache from T3.1.

**Depends on:** T3.2 (FQL lexer), T3.1 (SQLite schema — to know what virtual tables exist)
**Blocks:** T3.4 (terminal charts), T3.5 (FQL REPL command)

## Create `fql/tables.go` — Virtual Table Definitions

Define the virtual tables and their columns that FQL queries can access.

```go
package fql

// VirtualTable defines a table available to FQL queries.
type VirtualTable struct {
    Name    string
    Columns []VirtualColumn
    SQL     string // the SQL SELECT that produces this virtual table
}

// VirtualColumn defines a column in a virtual table.
type VirtualColumn struct {
    Name        string
    Type        string // "text", "real", "integer"
    Description string
}

// Tables is the registry of all available FQL virtual tables.
var Tables = map[string]*VirtualTable{
    "transactions": {
        Name: "transactions",
        Columns: []VirtualColumn{
            {Name: "id",          Type: "text"},
            {Name: "date",        Type: "text"},
            {Name: "description", Type: "text"},
            {Name: "account",     Type: "text"},
            {Name: "amount",      Type: "real"},
            {Name: "currency",    Type: "text"},
            {Name: "tags",        Type: "text"},
        },
        SQL: `
            SELECT
                t.id,
                t.date,
                t.description,
                t.status,
                p.account,
                p.amount,
                p.currency,
                COALESCE(
                    (SELECT GROUP_CONCAT(key || '=' || value, ';')
                     FROM transaction_tags WHERE transaction_id = t.id),
                    ''
                ) AS tags
            FROM transactions t
            JOIN postings p ON p.transaction_id = t.id
        `,
    },
    "accounts": {
        Name: "accounts",
        Columns: []VirtualColumn{
            {Name: "name",                Type: "text"},
            {Name: "type",                Type: "text"},
            {Name: "transaction_count",   Type: "integer"},
            {Name: "total_amount",        Type: "real"},
            {Name: "last_transaction",    Type: "text"},
            {Name: "first_transaction",   Type: "text"},
        },
        SQL: `
            SELECT
                p.account AS name,
                CASE
                    WHEN p.account LIKE 'assets%' THEN 'asset'
                    WHEN p.account LIKE 'liabilities%' THEN 'liability'
                    WHEN p.account LIKE 'equity%' THEN 'equity'
                    WHEN p.account LIKE 'income%' THEN 'income'
                    WHEN p.account LIKE 'expenses%' THEN 'expense'
                    ELSE 'other'
                END AS type,
                COUNT(*) AS transaction_count,
                SUM(p.amount) AS total_amount,
                MAX(t.date) AS last_transaction,
                MIN(t.date) AS first_transaction
            FROM postings p
            JOIN transactions t ON t.id = p.transaction_id
            GROUP BY p.account
        `,
    },
    "spending": {
        Name: "spending",
        Columns: []VirtualColumn{
            {Name: "date",              Type: "text"},
            {Name: "month",             Type: "text"},
            {Name: "year",              Type: "text"},
            {Name: "account",           Type: "text"},
            {Name: "transaction_count", Type: "integer"},
            {Name: "total_amount",      Type: "real"},
            {Name: "avg_amount",        Type: "real"},
            {Name: "min_amount",        Type: "real"},
            {Name: "max_amount",        Type: "real"},
        },
        SQL: `
            SELECT
                t.date,
                substr(t.date, 1, 7) AS month,
                substr(t.date, 1, 4) AS year,
                p.account,
                COUNT(*) AS transaction_count,
                SUM(p.amount) AS total_amount,
                AVG(p.amount) AS avg_amount,
                MIN(p.amount) AS min_amount,
                MAX(p.amount) AS max_amount
            FROM transactions t
            JOIN postings p ON p.transaction_id = t.id
            GROUP BY t.date, p.account
        `,
    },
}
```

## Create `fql/parser.go` — FQL AST and Parser

### Query AST types

```go
// Query is the top-level FQL query node.
type Query struct {
    Distinct   bool
    Columns    []SelectColumn
    From       string     // virtual table name
    Where      Expression // nil if no WHERE
    GroupBy    []string
    Having     Expression // nil if no HAVING
    OrderBy    []OrderSpec
    Limit      *int
    Offset     *int
}

// SelectColumn is a column in the SELECT list.
type SelectColumn struct {
    Expr  Expression
    Alias string // from AS clause, may be empty
    Star  bool   // true for SELECT *
}

// OrderSpec is one ORDER BY term.
type OrderSpec struct {
    Column string
    Desc   bool
}

// Expression is a WHERE/HAVING condition node.
type Expression interface{ exprNode() }

// Various expression types:
type BinaryExpr struct {
    Left  Expression
    Op    string // "=", "!=", "<", "<=", ">", ">=", "AND", "OR", "LIKE", "BETWEEN", "IN", "NOT IN"
    Right Expression
}

type UnaryExpr struct {
    Op    string // "NOT", "IS NULL", "IS NOT NULL"
    Expr  Expression
}

type Identifier struct { Name string }
type NumberLiteral struct { Value float64; Raw string }
type StringLiteral struct { Value string }
type StarLiteral struct{}  // for COUNT(*)

type FunctionCall struct {
    Name string   // COUNT, SUM, AVG, MIN, MAX
    Args []Expression
    Star bool     // true for COUNT(*)
}

type InExpr struct {
    Left   Expression
    Values []Expression
    Not    bool
}

type BetweenExpr struct {
    Expr  Expression
    Low   Expression
    High  Expression
}

type IsNullExpr struct {
    Expr Expression
    Not  bool
}
```

```go
type Parser struct {
    lexer *Lexer

func NewParser(input string) (*Parser, error) {
    l := NewLexer(input)
    if err := l.Tokenize(); err != nil {
        return nil, err
    }
    return &Parser{lexer: l}, nil
}
```

### Fixes for known bugs

#### Fix 1: COUNT(*) without alias
The old code required an alias after `COUNT(*)`. Remove that requirement.
When parsing a `SelectColumn`, alias is optional — only consume `AS alias` if `AS` follows.

#### Fix 2: GROUP BY with aggregates in SELECT
Allow `COUNT(*)`, `SUM(expr)`, etc. in SELECT without requiring an alias.
When no alias is given, generate one automatically in the compiler: `count_star`, `sum_amount`, etc.

#### Fix 3: HAVING with negative numbers
The HAVING expression is parsed the same way as WHERE. Since the FQL lexer now
correctly handles `-50` as a single negative number token in operator context,
HAVING just works.

#### Fix 4: Compound AND/OR with negative numbers
Same fix — the lexer handles `amount < -20` correctly, so the parser just sees
`Identifier`, `<`, `Number(-20)`.

## Create `fql/compiler.go` — FQL → SQL

```go
// Compiled is the result of FQL compilation.
type Compiled struct {
    SQL    string
    Params []interface{}
}

// Compiler translates a Query AST to parameterized SQLite SQL.
type Compiler struct {
    params []interface{}
}

// Compile validates and compiles a Query to SQL.
func (c *Compiler) Compile(q *Query) (*Compiled, error)
```

### Compilation strategy

1. **Validate table**: check `q.From` against `Tables` map → error if unknown
2. **Validate columns**: for each `SelectColumn`, check identifiers against the table's
   `VirtualColumn` list → error if unknown column name
3. **Build SQL**: wrap the virtual table's SQL in a subquery:
   ```sql
   SELECT <columns> FROM (<virtual_table_sql>) AS _vt
   [WHERE <conditions>]
   [GROUP BY <columns>]
   [HAVING <conditions>]
   [ORDER BY <columns>]
   [LIMIT n] [OFFSET n]
   ```
4. **Parameterize** all literal values → replace with `?` and append to `params`

### Column validation notes
- For `accounts` table, GROUP BY is built into the virtual table SQL; don't add another GROUP BY unless the user explicitly queries that
- Aggregate functions (`SUM`, `COUNT`, etc.) are always valid in SELECT
- For DISTINCT, add `SELECT DISTINCT ...`

## Create `fql/fql_test.go` — Integration Tests

Test all the cases from the spec (both the passing ones and the fixes):

```go
func TestBasicSelect(t *testing.T) {
    c, err := Compile("SELECT id, date, description, amount FROM transactions LIMIT 10")
    if err != nil { t.Fatal(err) }
    // verify SQL contains LIMIT 10
}

func TestNegativeNumber(t *testing.T) {
    // Bug fix: was failing before
    c, err := Compile("SELECT id, date, amount FROM transactions WHERE amount < -100")
    if err != nil { t.Fatalf("expected success, got error: %v", err) }
    // verify params contains -100.0
}

func TestCountStarNoAlias(t *testing.T) {
    // Bug fix: was requiring alias
    c, err := Compile("SELECT owner_name, COUNT(*) FROM transactions GROUP BY owner_name")
    if err != nil { t.Fatalf("expected success, got error: %v", err) }
}

func TestHavingNegative(t *testing.T) {
    // Bug fix: HAVING total < -50
    c, err := Compile("SELECT category, SUM(amount) AS total FROM transactions GROUP BY category HAVING total < -50")
    if err != nil { t.Fatalf("expected success, got error: %v", err) }
}

func TestCompoundOrWithNegative(t *testing.T) {
    // Bug fix: (category = 'food' OR category = 'transport') AND amount < -20
    c, err := Compile("SELECT id, amount FROM transactions WHERE (category = 'food' OR category = 'transport') AND amount < -20")
    if err != nil { t.Fatalf("expected success, got error: %v", err) }
}

func TestUnknownTable(t *testing.T) {
    _, err := Compile("SELECT id FROM real_transactions")
    if err == nil { t.Fatal("expected error for unknown table") }
}

func TestUnknownColumn(t *testing.T) {
    _, err := Compile("SELECT secret_column FROM transactions")
    if err == nil { t.Fatal("expected error for unknown column") }
}

func TestMissingFrom(t *testing.T) {
    _, err := Compile("SELECT id WHERE amount > 0")
    if err == nil { t.Fatal("expected error for missing FROM") }
}

func TestInList(t *testing.T) {
    c, err := Compile("SELECT id FROM transactions WHERE account IN ('food', 'transport')")
    if err != nil { t.Fatal(err) }
    if len(c.Params) != 2 { t.Fatalf("expected 2 params, got %d", len(c.Params)) }
}

func TestDateRange(t *testing.T) {
    c, err := Compile("SELECT date, amount FROM transactions WHERE date BETWEEN '2024-01-01' AND '2024-12-31'")
    if err != nil { t.Fatal(err) }
}
```
